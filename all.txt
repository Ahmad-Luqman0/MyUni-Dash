-- 01. Dashboard_MyUni_Filters: the filter lists shown above every tab, and the semester strip. One call per list.
-- @Section = Companies | Branches | Faculties | Departments | Degrees | Semesters | EmployeeDepartments
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Filters;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Filters
    @Section      VARCHAR(30),               -- which list: Companies, Branches, Faculties, Departments, Degrees, Semesters, EmployeeDepartments
    @CompanyId    INT              = NULL,   -- every list except Companies: the chosen company
    @BranchId     INT              = NULL,   -- Degrees, Semesters, EmployeeDepartments: NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- Departments, Degrees: NULL = all
    @DepartmentId INT              = NULL    -- Degrees: NULL or 0 = all
AS
BEGIN
    SET NOCOUNT ON;

    -- 1. Companies: every active company
    IF @Section = 'Companies'
    BEGIN
        SELECT C.Id, C.CompanyName
        FROM Company C WITH (NOLOCK)
        WHERE C.Status = 1
        ORDER BY C.Id;

        RETURN;
    END

    -- 2. Branches: active branches (campuses) of the company.
    --    Students = active students of the branch, so the web can tell which campuses are in use.
    IF @Section = 'Branches'
    BEGIN
        SELECT B.Id, B.[Description], B.Code,
               CT.[Description] AS City,
               (SELECT COUNT(*) FROM UNIStudents S WITH (NOLOCK) WHERE S.BranchId = B.Id AND S.Status = 1) AS Students
        FROM Branches B WITH (NOLOCK)
        LEFT JOIN City CT WITH (NOLOCK) ON CT.Id = B.CityId
        WHERE B.Status = 1 AND B.CompanyId = @CompanyId
        ORDER BY B.Id;

        RETURN;
    END

    -- 3. Faculties: active faculties of the company
    IF @Section = 'Faculties'
    BEGIN
        SELECT F.FacultyGUID AS Id, F.[Description], F.Code
        FROM Faculty F WITH (NOLOCK)
        WHERE F.Status = 1 AND F.CompanyId = @CompanyId
        ORDER BY F.[Description];

        RETURN;
    END

    -- 4. Departments: departments that belong to a faculty of the company
    IF @Section = 'Departments'
    BEGIN
        SELECT DISTINCT D.Id, D.[Description], F.FacultyGUID AS FacultyId, F.[Description] AS Faculty
        FROM FacultyDetails FD WITH (NOLOCK)
        JOIN Faculty F     WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
        JOIN Departments D WITH (NOLOCK) ON D.Id = FD.DepartmentId
        WHERE F.Status = 1 AND F.CompanyId = @CompanyId
          AND (@FacultyId IS NULL OR FD.FacultyGUID = @FacultyId)
        ORDER BY D.[Description];

        RETURN;
    END

    -- 5. Degrees: active degree programs
    IF @Section = 'Degrees'
    BEGIN
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.DepartmentId,
               DL.DegreeLevelName AS DegreeLevel
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN UNIDegreeLevels DL ON DL.Id = UD.DegreeLevel
        WHERE UD.Status = 1 AND UD.CompanyId = @CompanyId
          AND (ISNULL(@BranchId, 0) = 0 OR UD.BranchId = @BranchId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD WITH (NOLOCK)
                                             WHERE FD.DepartmentId = UD.DepartmentId AND FD.FacultyGUID = @FacultyId))
        ORDER BY UD.DegreeName;

        RETURN;
    END

    -- 6. Semesters: semesters of the company's academic calendars.
    --    Students = active students registered in the semester (RegisterInId, or the first semester when RegisterInId is empty).
    IF @Section = 'Semesters'
    BEGIN
        SELECT SEM.Id, SEM.Code, AC.CalenderName AS Calendar, SEM.StartDate, SEM.EndDate,
               ISNULL(SEM.IsAdmissionOpen, 0) AS IsAdmissionOpen,
               (SELECT COUNT(DISTINCT R.StudentsId)
                FROM UNIStudentsRegistrations R WITH (NOLOCK)
                JOIN UNIStudents S WITH (NOLOCK) ON S.Id = R.StudentsId AND S.Status = 1
                WHERE ISNULL(R.RegisterInId, R.SemesterSessionId) = SEM.Id
                  AND R.Status = 1 AND R.CompanyId = @CompanyId
                  AND (ISNULL(@BranchId, 0) = 0 OR R.BranchId = @BranchId)) AS Students
        FROM UNIAcademicYearSemesterDetails SEM WITH (NOLOCK)
        JOIN UNIAcademicCalender AC WITH (NOLOCK) ON AC.Id = SEM.AcademicCalenderId
        WHERE AC.CompanyId = @CompanyId
          AND (ISNULL(@BranchId, 0) = 0 OR AC.BranchId = @BranchId)
        ORDER BY SEM.StartDate, SEM.OrderId;

        RETURN;
    END

    -- 7. EmployeeDepartments: departments that have active employees (the Department filter of the HR tabs).
    IF @Section = 'EmployeeDepartments'
    BEGIN
        SELECT D.Id, D.[Description], COUNT(*) AS Employees
        FROM Employees E WITH (NOLOCK)
        JOIN Departments D WITH (NOLOCK) ON D.Id = E.DepartmentId
        WHERE E.CompanyId = @CompanyId AND E.Status = 1
          AND (ISNULL(@BranchId, 0) = 0 OR E.BranchId = @BranchId)
        GROUP BY D.Id, D.[Description]
        ORDER BY D.[Description];

        RETURN;
    END
END
GO

-- 02. Dashboard_MyUni_StudentSummary: the headline numbers of the Overview tab, one row.
-- Every figure follows @StartDate / @EndDate:
-- Students: students on record in the allowed degrees; with dates, those registered in a semester running in the dates
--           (a semester runs from its start until the next one starts, so every date belongs to a semester).
-- RegisteredStudents, NewAdmissions: for the chosen semester, or the semesters running in the dates.
-- DegreeCompleted: students flagged as completed; with dates, those whose completion date falls in them.
-- Billed, Received, Outstanding, Discounts: challans dated in the range (rules of Dashboard_MyUni_Receivables).
-- Overdue, Defaulters: challans that fell due in the dates, judged on @EndDate (rules of Dashboard_MyUni_Defaulters).
-- Collected: receipts dated in the range (rules of Dashboard_MyUni_Collections).
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_StudentSummary;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_StudentSummary
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL,   -- whole day included; NULL = no limit
    @RequireApproved BIT           = 1       -- not used any more (kept so the API's call still works)
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From     DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To       DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));
    DECLARE @AllDates BIT  = CASE WHEN @StartDate IS NULL AND @EndDate IS NULL THEN 1 ELSE 0 END;
    DECLARE @Today    DATE = CAST(GETDATE() AS DATE);
    DECLARE @AsOf     DATE = CASE WHEN @EndDate IS NULL OR @EndDate > @Today THEN @Today ELSE DATEADD(DAY, 1, @EndDate) END;

    -- The challans that already have a verified bank receipt, keyed so each cash fee receipt is checked with one
    -- index seek. (Checking against the bank receipt tables directly took minutes on a full database.)
    DECLARE @BankPaid TABLE (StudentId UNIQUEIDENTIFIER NOT NULL, ChallanNo NVARCHAR(200) NOT NULL, PRIMARY KEY (StudentId, ChallanNo));
    INSERT INTO @BankPaid (StudentId, ChallanNo)
    SELECT DISTINCT BD.StudentId, CAST(BD.ChallanNo AS NVARCHAR(200))
    FROM UNIBankReceiptsDetail BD WITH (NOLOCK)
    JOIN UNIBankReceipts BH WITH (NOLOCK) ON BH.Id = BD.BankReceiptId
    WHERE BH.Status = 1 AND BD.DocStatus = 35 AND BH.BankReceiptStatus IN (2, 34)
      AND BD.StudentId IS NOT NULL AND BD.ChallanNo IS NOT NULL;

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    ),
    -- Sem: the semesters the dates select. A semester is taken to run from its start until the next semester of the
    -- company starts (or to its own end date when that is later), and the latest one has no end. So the gaps between
    -- semesters, and a semester stored with a one-day span, still belong to a semester.
    Sem AS (
        SELECT SD.Id
        FROM UNIAcademicYearSemesterDetails SD WITH (NOLOCK)
        JOIN UNIAcademicCalender AC WITH (NOLOCK) ON AC.Id = SD.AcademicCalenderId AND AC.CompanyId = @CompanyId
        OUTER APPLY (SELECT MIN(N.StartDate) AS NextStart
                     FROM UNIAcademicYearSemesterDetails N WITH (NOLOCK)
                     JOIN UNIAcademicCalender NC WITH (NOLOCK) ON NC.Id = N.AcademicCalenderId AND NC.CompanyId = @CompanyId
                     WHERE N.StartDate > SD.StartDate) NX
        WHERE SD.StartDate < @To
          AND CASE WHEN NX.NextStart IS NULL THEN CAST('99991231' AS DATE)
                   WHEN SD.EndDate > DATEADD(DAY, -1, NX.NextStart) THEN SD.EndDate
                   ELSE DATEADD(DAY, -1, NX.NextStart) END >= @From
    ),
    Ch AS (
        SELECT C.Id, C.StudentId, C.DegreeId, C.ChallanDescription,
               ISNULL(C.ValidityDate, C.DueDate) AS DueOn,
               ISNULL(C.NetChallanAmt, ISNULL(C.TotalAmount, 0) - ISNULL(C.TotalDiscountAmt, 0)) AS Net,
               ISNULL(C.TotalDiscountAmt, 0) + ISNULL(C.GovtFeeDiscountAmt, 0) AS Discount,
               CASE WHEN C.TransDate >= @From AND C.TransDate < @To THEN 1 ELSE 0 END AS InRange,
               CASE WHEN C.ParentIdGuid IS NULL AND C.ChallanStatus NOT IN (3, 40) THEN 1 ELSE 0 END AS IsBilled,
               CASE WHEN C.ChallanStatus IN (1, 62) THEN 1 ELSE 0 END AS IsPaid,
               CASE WHEN C.ParentIdGuid IS NULL AND C.ChallanStatus NOT IN (1, 62, 3, 40, 63, 64)
                         AND ISNULL(C.ValidityDate, C.DueDate) < @AsOf THEN 1 ELSE 0 END AS IsOverdue,
               CASE WHEN @AllDates = 1 OR (ISNULL(C.ValidityDate, C.DueDate) >= @From AND ISNULL(C.ValidityDate, C.DueDate) < @To) THEN 1 ELSE 0 END AS InDue
        FROM UNIStudentsChallan C WITH (NOLOCK)
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = C.StudentId AND S.Status = 1
        JOIN Deg G ON G.Id = C.DegreeId
        WHERE C.CompanyId = @CompanyId
          AND C.Status = 1
          AND (ISNULL(@BranchId, 0) = 0 OR C.BranchId = @BranchId)
          AND (@SemesterId IS NULL OR C.RegisterInId = @SemesterId)
    ),
    -- Due: one row per student and degree; a defaulter has a balance and at least one overdue challan.
    Due AS (
        SELECT C.StudentId, C.DegreeId,
               SUM(C.IsBilled * C.Net) AS TotalFee,
               SUM(C.IsPaid * C.Net)   AS Received,
               SUM(C.IsOverdue)        AS UnpaidChallans,
               MIN(CASE WHEN C.IsOverdue = 1 THEN C.DueOn END) AS OldestDueDate
        FROM Ch C
        WHERE C.InDue = 1
        GROUP BY C.StudentId, C.DegreeId
        HAVING SUM(C.IsBilled * C.Net) - SUM(C.IsPaid * C.Net) > 0 AND SUM(C.IsOverdue) > 0
    ),
    Rec AS (
        SELECT ISNULL(SC.DegreeId, S.DegreeId) AS DegreeId, ISNULL(DT.Amount, ISNULL(FR.TotalAmount, 0)) AS Amount
        FROM AEOFeeReceipts FR WITH (NOLOCK)
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = FR.UNIStudentId AND S.Status = 1
        LEFT JOIN UNIStudentsChallan SC WITH (NOLOCK) ON SC.Id = FR.UNIChallanId
        OUTER APPLY (SELECT SUM(ISNULL(FRD.Amount, 0)) AS Amount
                     FROM AEOFeeReceiptsDetail FRD WITH (NOLOCK) WHERE FRD.FeeReceiptId = FR.Id) DT
        WHERE FR.CompanyId = @CompanyId
          AND FR.Status = 1
          AND FR.FeeStatus IN (4, 19)                       -- 4 = received (set when the receipt is saved), 19 = approved
          AND (SC.Id IS NULL OR SC.ChallanStatus <> 40)     -- not against a cancelled challan
          -- not already counted as a bank receipt for the same challan
          AND NOT EXISTS (SELECT 1 FROM @BankPaid BP WHERE BP.StudentId = FR.UNIStudentId AND BP.ChallanNo = SC.ChallanNo)
          AND (ISNULL(@BranchId, 0) = 0 OR FR.BranchId = @BranchId)
          AND FR.TransDate >= @From AND FR.TransDate < @To
          AND (@SemesterId IS NULL OR SC.RegisterInId = @SemesterId)

        UNION ALL

        SELECT ISNULL(BRD.DegreeId, S.DegreeId), ISNULL(BRD.Amount, 0)
        FROM UNIBankReceiptsDetail BRD WITH (NOLOCK)
        JOIN UNIBankReceipts BR WITH (NOLOCK) ON BR.Id = BRD.BankReceiptId
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = BRD.StudentId AND S.Status = 1
        OUTER APPLY (SELECT TOP 1 C.RegisterInId
                     FROM UNIStudentsChallan C WITH (NOLOCK)
                     WHERE C.StudentId = BRD.StudentId AND C.ChallanNo = BRD.ChallanNo
                     ORDER BY C.TransDate DESC) SC
        WHERE BR.Status = 1
          AND BRD.DocStatus = 35
          AND BR.BankReceiptStatus IN (2, 34)
          AND (ISNULL(@BranchId, 0) = 0 OR BR.BranchId = @BranchId)
          AND BR.UploadedDate >= @From AND BR.UploadedDate < @To
          AND (@SemesterId IS NULL OR SC.RegisterInId = @SemesterId)
    )
    SELECT
        (SELECT COUNT(*) FROM UNIStudents S WITH (NOLOCK) JOIN Deg G ON G.Id = S.DegreeId
         WHERE S.Status = 1 AND (ISNULL(@BranchId, 0) = 0 OR S.BranchId = @BranchId)
           AND (@AllDates = 1 OR EXISTS (SELECT 1 FROM UNIStudentsRegistrations R WITH (NOLOCK)
                                         WHERE R.StudentsId = S.Id AND R.Status = 1
                                           AND ISNULL(R.RegisterInId, R.SemesterSessionId) IN (SELECT Id FROM Sem)))) AS Students,

        (SELECT COUNT(DISTINCT R.StudentsId)
         FROM UNIStudentsRegistrations R WITH (NOLOCK)
         JOIN UNIStudents S WITH (NOLOCK) ON S.Id = R.StudentsId AND S.Status = 1
         JOIN Deg G ON G.Id = R.DegreeId
         WHERE R.CompanyId = @CompanyId AND R.Status = 1
           AND (ISNULL(@BranchId, 0) = 0 OR R.BranchId = @BranchId)
           AND (@SemesterId IS NULL OR ISNULL(R.RegisterInId, R.SemesterSessionId) = @SemesterId)
           AND (@AllDates = 1 OR ISNULL(R.RegisterInId, R.SemesterSessionId) IN (SELECT Id FROM Sem))) AS RegisteredStudents,

        (SELECT COUNT(*) FROM UNIStudents S WITH (NOLOCK) JOIN Deg G ON G.Id = S.DegreeId
         WHERE S.Status = 1 AND S.IsDegreeCompleted = 1
           AND (ISNULL(@BranchId, 0) = 0 OR S.BranchId = @BranchId)
           AND (@AllDates = 1 OR (S.CompletionDate >= @From AND S.CompletionDate < @To))) AS DegreeCompleted,

        (SELECT COUNT(*) FROM Deg G WHERE G.Status = 1) AS Programs,

        (SELECT COUNT(DISTINCT SA.StudentsId)
         FROM UNIStudentAnnounceDetails SA WITH (NOLOCK)
         JOIN UNIStudents S WITH (NOLOCK) ON S.Id = SA.StudentsId AND S.Status = 1
         JOIN Deg G ON G.Id = SA.DegreeId
         WHERE SA.Status = 1
           AND (ISNULL(@BranchId, 0) = 0 OR S.BranchId = @BranchId)
           AND (@SemesterId IS NULL OR SA.SemesterSessionId = @SemesterId)
           AND (@AllDates = 1 OR SA.SemesterSessionId IN (SELECT Id FROM Sem))) AS NewAdmissions,

        ISNULL((SELECT SUM(C.IsBilled * C.Net) FROM Ch C WHERE C.InRange = 1), 0) AS Billed,
        ISNULL((SELECT SUM(C.IsPaid * C.Net)   FROM Ch C WHERE C.InRange = 1), 0) AS Received,
        ISNULL((SELECT SUM(C.IsBilled * C.Net) - SUM(C.IsPaid * C.Net) FROM Ch C WHERE C.InRange = 1), 0) AS Outstanding,
        ISNULL((SELECT SUM(C.Net) FROM Ch C WHERE C.IsOverdue = 1 AND C.InDue = 1), 0) AS Overdue,
        (SELECT COUNT(DISTINCT D.StudentId) FROM Due D) AS Defaulters,
        ISNULL((SELECT SUM(C.IsBilled * C.Discount) FROM Ch C WHERE C.InRange = 1), 0) AS Discounts,
        ISNULL((SELECT SUM(R.Amount) FROM Rec R JOIN Deg G ON G.Id = R.DegreeId), 0) AS Collected;
END

-- 03. Dashboard_MyUni_Registrations: registered students per semester, degree, semester number and shift (Students tab).
-- A registration counts when it is active, its student is active, and it falls in the semester
-- (RegisterInId; the first semester, SemesterSessionId, when RegisterInId is empty).
-- With dates chosen, only the semesters running in @StartDate to @EndDate are counted. A semester runs from its
-- start until the next one starts, so every date belongs to a semester.
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Registrations;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Registrations
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From     DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To       DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));
    DECLARE @AllDates BIT  = CASE WHEN @StartDate IS NULL AND @EndDate IS NULL THEN 1 ELSE 0 END;

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    ),
    -- Sem: the semesters the dates select. A semester is taken to run from its start until the next semester of the
    -- company starts (or to its own end date when that is later), and the latest one has no end. So the gaps between
    -- semesters, and a semester stored with a one-day span, still belong to a semester.
    Sem AS (
        SELECT SD.Id
        FROM UNIAcademicYearSemesterDetails SD WITH (NOLOCK)
        JOIN UNIAcademicCalender AC WITH (NOLOCK) ON AC.Id = SD.AcademicCalenderId AND AC.CompanyId = @CompanyId
        OUTER APPLY (SELECT MIN(N.StartDate) AS NextStart
                     FROM UNIAcademicYearSemesterDetails N WITH (NOLOCK)
                     JOIN UNIAcademicCalender NC WITH (NOLOCK) ON NC.Id = N.AcademicCalenderId AND NC.CompanyId = @CompanyId
                     WHERE N.StartDate > SD.StartDate) NX
        WHERE SD.StartDate < @To
          AND CASE WHEN NX.NextStart IS NULL THEN CAST('99991231' AS DATE)
                   WHEN SD.EndDate > DATEADD(DAY, -1, NX.NextStart) THEN SD.EndDate
                   ELSE DATEADD(DAY, -1, NX.NextStart) END >= @From
    )
    SELECT SEM.Id AS SemesterId, SEM.Code AS Semester, SEM.StartDate AS SemesterStart,
           G.Id AS DegreeId, G.DegreeCode, G.DegreeName AS Degree, G.Department, G.Faculty,
           R.SemesterNo, PT.[Description] AS ProgramType,
           COUNT(DISTINCT R.StudentsId) AS Students
    FROM UNIStudentsRegistrations R WITH (NOLOCK)
    JOIN UNIStudents S WITH (NOLOCK) ON S.Id = R.StudentsId AND S.Status = 1
    JOIN Deg G ON G.Id = R.DegreeId
    LEFT JOIN UNIAcademicYearSemesterDetails SEM WITH (NOLOCK) ON SEM.Id = ISNULL(R.RegisterInId, R.SemesterSessionId)
    LEFT JOIN ProgramType PT WITH (NOLOCK) ON PT.Id = S.ShiftId
    WHERE R.CompanyId = @CompanyId
      AND R.Status = 1
      AND (ISNULL(@BranchId, 0) = 0 OR R.BranchId = @BranchId)
      AND (@SemesterId IS NULL OR ISNULL(R.RegisterInId, R.SemesterSessionId) = @SemesterId)
      AND (@AllDates = 1 OR ISNULL(R.RegisterInId, R.SemesterSessionId) IN (SELECT Id FROM Sem))
    GROUP BY SEM.Id, SEM.Code, SEM.StartDate, G.Id, G.DegreeCode, G.DegreeName, G.Department, G.Faculty,
             R.SemesterNo, PT.[Description]
    ORDER BY SEM.StartDate, G.DegreeName, R.SemesterNo;
END

-- 05. Dashboard_MyUni_Collections: money received per day, source, degree and challan type (Collections tab).
-- Two sources:
--   Cash Fee Receipt = AEOFeeReceipts of a university student, not deleted (Status 1), received or approved
--                      (FeeStatus 4 or 19), dated by TransDate. The ERP saves a university fee receipt with FeeStatus 4
--                      and marks its challan paid at once (UNIFeeRecipts_Insertion), so 4 is money received.
--                      Left out: receipts against a cancelled challan, as in the ERP's list AEOFeeReceipts_GetListForUNI,
--                      and receipts whose challan already has a verified bank receipt (so nothing is counted twice).
--   Bank receipt     = UNIBankReceiptsDetail verified (DocStatus 35) on an approved / posted upload (BankReceiptStatus 2, 34), dated by UploadedDate
-- @RequireApproved is no longer used (kept so the API's call still works).
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Collections;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Collections
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL,   -- whole day included; NULL = no limit
    @RequireApproved BIT           = 1       -- not used any more
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To   DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));

    -- The challans that already have a verified bank receipt, keyed so each cash fee receipt is checked with one
    -- index seek. (Checking against the bank receipt tables directly took minutes on a full database.)
    DECLARE @BankPaid TABLE (StudentId UNIQUEIDENTIFIER NOT NULL, ChallanNo NVARCHAR(200) NOT NULL, PRIMARY KEY (StudentId, ChallanNo));
    INSERT INTO @BankPaid (StudentId, ChallanNo)
    SELECT DISTINCT BD.StudentId, CAST(BD.ChallanNo AS NVARCHAR(200))
    FROM UNIBankReceiptsDetail BD WITH (NOLOCK)
    JOIN UNIBankReceipts BH WITH (NOLOCK) ON BH.Id = BD.BankReceiptId
    WHERE BH.Status = 1 AND BD.DocStatus = 35 AND BH.BankReceiptStatus IN (2, 34)
      AND BD.StudentId IS NOT NULL AND BD.ChallanNo IS NOT NULL;

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    ),
    Rec AS (
        SELECT CAST(FR.TransDate AS DATE) AS ReceiptDate, CAST('Cash Fee Receipt' AS VARCHAR(20)) AS Source,
               ISNULL(SC.DegreeId, S.DegreeId) AS DegreeId, SC.ChallanTypeId,
               ISNULL(DT.Amount, ISNULL(FR.TotalAmount, 0)) AS Amount
        FROM AEOFeeReceipts FR WITH (NOLOCK)
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = FR.UNIStudentId AND S.Status = 1
        LEFT JOIN UNIStudentsChallan SC WITH (NOLOCK) ON SC.Id = FR.UNIChallanId
        OUTER APPLY (SELECT SUM(ISNULL(FRD.Amount, 0)) AS Amount
                     FROM AEOFeeReceiptsDetail FRD WITH (NOLOCK) WHERE FRD.FeeReceiptId = FR.Id) DT
        WHERE FR.CompanyId = @CompanyId
          AND FR.Status = 1
          AND FR.FeeStatus IN (4, 19)                       -- 4 = received (set when the receipt is saved), 19 = approved
          AND (SC.Id IS NULL OR SC.ChallanStatus <> 40)     -- not against a cancelled challan
          -- not already counted as a bank receipt for the same challan
          AND NOT EXISTS (SELECT 1 FROM @BankPaid BP WHERE BP.StudentId = FR.UNIStudentId AND BP.ChallanNo = SC.ChallanNo)
          AND (ISNULL(@BranchId, 0) = 0 OR FR.BranchId = @BranchId)
          AND FR.TransDate >= @From AND FR.TransDate < @To
          AND (@SemesterId IS NULL OR SC.RegisterInId = @SemesterId)

        UNION ALL

        SELECT BR.UploadedDate, CAST('Bank receipt' AS VARCHAR(20)),
               ISNULL(BRD.DegreeId, S.DegreeId), SC.ChallanTypeId,
               ISNULL(BRD.Amount, 0)
        FROM UNIBankReceiptsDetail BRD WITH (NOLOCK)
        JOIN UNIBankReceipts BR WITH (NOLOCK) ON BR.Id = BRD.BankReceiptId
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = BRD.StudentId AND S.Status = 1
        OUTER APPLY (SELECT TOP 1 C.ChallanTypeId, C.RegisterInId
                     FROM UNIStudentsChallan C WITH (NOLOCK)
                     WHERE C.StudentId = BRD.StudentId AND C.ChallanNo = BRD.ChallanNo
                     ORDER BY C.TransDate DESC) SC
        WHERE BR.Status = 1
          AND BRD.DocStatus = 35
          AND BR.BankReceiptStatus IN (2, 34)
          AND (ISNULL(@BranchId, 0) = 0 OR BR.BranchId = @BranchId)
          AND BR.UploadedDate >= @From AND BR.UploadedDate < @To
          AND (@SemesterId IS NULL OR SC.RegisterInId = @SemesterId)
    )
    SELECT R.ReceiptDate, R.Source,
           G.Id AS DegreeId, G.DegreeCode, G.DegreeName AS Degree, G.Department, G.Faculty,
           CT.ChallanType,
           COUNT(*)      AS Receipts,
           SUM(R.Amount) AS Amount
    FROM Rec R
    JOIN Deg G ON G.Id = R.DegreeId
    LEFT JOIN UNIChallanTypes CT ON CT.Id = R.ChallanTypeId
    GROUP BY R.ReceiptDate, R.Source, G.Id, G.DegreeCode, G.DegreeName, G.Department, G.Faculty, CT.ChallanType
    ORDER BY R.ReceiptDate, R.Source, G.DegreeName;
END

-- 06. Dashboard_MyUni_Receivables: fee billed, received and overdue per semester, degree and challan type (Receivables tab).
-- Same rules as the ERP report UNIStudents_Rp_GetDefaulterData:
--   Billed   = NetChallanAmt of main challans (ParentIdGuid empty) that are not expired / cancelled (ChallanStatus 3, 40)
--   Received = NetChallanAmt of paid challans (ChallanStatus 1, 62)
--   Overdue  = main challans still open (not 1, 62, 3, 40, 63, 64) whose validity date (else due date) has passed
-- Balance = Billed - Received. The date range is on the challan date (TransDate).
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Receivables;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Receivables
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From  DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To    DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));
    DECLARE @Today DATE = CAST(GETDATE() AS DATE);

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    ),
    Ch AS (
        SELECT ISNULL(C.RegisterInId, C.SemesterSessionId) AS SemesterId, C.DegreeId, C.ChallanTypeId,
               ISNULL(C.NetChallanAmt, ISNULL(C.TotalAmount, 0) - ISNULL(C.TotalDiscountAmt, 0)) AS Net,
               ISNULL(C.TotalDiscountAmt, 0) + ISNULL(C.GovtFeeDiscountAmt, 0) AS Discount,
               ISNULL(C.FineAmount, 0) AS Fine,
               CASE WHEN C.ParentIdGuid IS NULL AND C.ChallanStatus NOT IN (3, 40) THEN 1 ELSE 0 END AS IsBilled,
               CASE WHEN C.ChallanStatus IN (1, 62) THEN 1 ELSE 0 END AS IsPaid,
               CASE WHEN C.ParentIdGuid IS NULL AND C.ChallanStatus NOT IN (1, 62, 3, 40, 63, 64)
                         AND ISNULL(C.ValidityDate, C.DueDate) < @Today THEN 1 ELSE 0 END AS IsOverdue
        FROM UNIStudentsChallan C WITH (NOLOCK)
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = C.StudentId AND S.Status = 1
        WHERE C.CompanyId = @CompanyId
          AND C.Status = 1
          AND (ISNULL(@BranchId, 0) = 0 OR C.BranchId = @BranchId)
          AND (@SemesterId IS NULL OR C.RegisterInId = @SemesterId)
          AND C.TransDate >= @From AND C.TransDate < @To
    )
    SELECT SEM.Id AS SemesterId, SEM.Code AS Semester, SEM.StartDate AS SemesterStart,
           G.Id AS DegreeId, G.DegreeCode, G.DegreeName AS Degree, G.Department, G.Faculty,
           CT.ChallanType,
           SUM(C.IsBilled)                 AS Challans,
           SUM(C.IsBilled * C.Net)         AS Billed,
           SUM(C.IsPaid * C.Net)           AS Received,
           SUM(C.IsOverdue)                AS OverdueChallans,
           SUM(C.IsOverdue * C.Net)        AS Overdue,
           SUM(C.IsBilled * C.Discount)    AS Discount,
           SUM(C.IsBilled * C.Fine)        AS Fine
    FROM Ch C
    JOIN Deg G ON G.Id = C.DegreeId
    LEFT JOIN UNIAcademicYearSemesterDetails SEM WITH (NOLOCK) ON SEM.Id = C.SemesterId
    LEFT JOIN UNIChallanTypes CT ON CT.Id = C.ChallanTypeId
    WHERE C.IsBilled = 1 OR C.IsPaid = 1
    GROUP BY SEM.Id, SEM.Code, SEM.StartDate, G.Id, G.DegreeCode, G.DegreeName, G.Department, G.Faculty, CT.ChallanType
    ORDER BY SEM.StartDate, G.DegreeName, CT.ChallanType;
END
GO

-- 07. Dashboard_MyUni_Defaulters: students with an overdue fee balance, one row per student and degree (Defaulters tab).
-- Rules of the ERP report UNIStudents_Rp_GetDefaulterData:
--   TotalFee = NetChallanAmt of main challans (ParentIdGuid empty) not expired / cancelled (ChallanStatus 3, 40)
--   Received = NetChallanAmt of paid challans (ChallanStatus 1, 62);  Balance = TotalFee - Received, must be above 0
--   and the student has a main challan still open (not 1, 62, 3, 40, 63, 64) whose validity date (else due date)
--   has passed. With dates chosen, only challans that fell due between @StartDate and @EndDate are looked at, and
--   overdue is judged on @EndDate (never on a day that has not come yet). With no dates: every challan, as of today.
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Defaulters;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Defaulters
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From     DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To       DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));
    DECLARE @AllDates BIT  = CASE WHEN @StartDate IS NULL AND @EndDate IS NULL THEN 1 ELSE 0 END;
    DECLARE @Today    DATE = CAST(GETDATE() AS DATE);
    DECLARE @AsOf     DATE = CASE WHEN @EndDate IS NULL OR @EndDate > @Today THEN @Today ELSE DATEADD(DAY, 1, @EndDate) END;

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    ),
    Ch AS (
        SELECT C.Id, C.StudentId, C.DegreeId, C.ChallanDescription,
               ISNULL(C.ValidityDate, C.DueDate) AS DueOn,
               ISNULL(C.NetChallanAmt, ISNULL(C.TotalAmount, 0) - ISNULL(C.TotalDiscountAmt, 0)) AS Net,
               ISNULL(C.TotalDiscountAmt, 0) + ISNULL(C.GovtFeeDiscountAmt, 0) AS Discount,
               CASE WHEN C.TransDate >= @From AND C.TransDate < @To THEN 1 ELSE 0 END AS InRange,
               CASE WHEN C.ParentIdGuid IS NULL AND C.ChallanStatus NOT IN (3, 40) THEN 1 ELSE 0 END AS IsBilled,
               CASE WHEN C.ChallanStatus IN (1, 62) THEN 1 ELSE 0 END AS IsPaid,
               CASE WHEN C.ParentIdGuid IS NULL AND C.ChallanStatus NOT IN (1, 62, 3, 40, 63, 64)
                         AND ISNULL(C.ValidityDate, C.DueDate) < @AsOf THEN 1 ELSE 0 END AS IsOverdue
        FROM UNIStudentsChallan C WITH (NOLOCK)
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = C.StudentId AND S.Status = 1
        JOIN Deg G ON G.Id = C.DegreeId
        WHERE C.CompanyId = @CompanyId
          AND C.Status = 1
          AND (ISNULL(@BranchId, 0) = 0 OR C.BranchId = @BranchId)
          AND (@SemesterId IS NULL OR C.RegisterInId = @SemesterId)
          AND (@AllDates = 1 OR (ISNULL(C.ValidityDate, C.DueDate) >= @From AND ISNULL(C.ValidityDate, C.DueDate) < @To))
    ),
    -- Due: one row per student and degree; a defaulter has a balance and at least one overdue challan.
    Due AS (
        SELECT C.StudentId, C.DegreeId,
               SUM(C.IsBilled * C.Net) AS TotalFee,
               SUM(C.IsPaid * C.Net)   AS Received,
               SUM(C.IsOverdue)        AS UnpaidChallans,
               MIN(CASE WHEN C.IsOverdue = 1 THEN C.DueOn END) AS OldestDueDate
        FROM Ch C
        GROUP BY C.StudentId, C.DegreeId
        HAVING SUM(C.IsBilled * C.Net) - SUM(C.IsPaid * C.Net) > 0 AND SUM(C.IsOverdue) > 0
    )
    SELECT D.StudentId, S.AdmissionNumber AS RegNo, S.StudentFullName AS Student,
           G.DegreeCode, G.DegreeName AS Degree, G.Department, G.Faculty,
           PT.[Description] AS ProgramType,
           D.TotalFee, D.Received, D.TotalFee - D.Received AS Balance,
           D.UnpaidChallans, D.OldestDueDate,
           DATEDIFF(DAY, D.OldestDueDate, CASE WHEN @EndDate IS NULL OR @EndDate > @Today THEN @Today ELSE @EndDate END) AS DaysOverdue,
           STUFF((SELECT ', ' + ISNULL(NULLIF(C.ChallanDescription, ''), 'Full fee')
                  FROM Ch C
                  WHERE C.StudentId = D.StudentId AND C.DegreeId = D.DegreeId AND C.IsOverdue = 1
                  ORDER BY C.DueOn
                  FOR XML PATH(''), TYPE).value('.', 'NVARCHAR(MAX)'), 1, 2, '') AS Remarks
    FROM Due D
    JOIN UNIStudents S WITH (NOLOCK) ON S.Id = D.StudentId
    JOIN Deg G ON G.Id = D.DegreeId
    LEFT JOIN ProgramType PT WITH (NOLOCK) ON PT.Id = S.ShiftId
    ORDER BY D.TotalFee - D.Received DESC, S.StudentFullName;
END
GO

-- 08. Dashboard_MyUni_Installments: installments per month, degree, plan status and state (Installments tab).
-- In the ERP an installment is a challan of its own: a child of the full-fee challan (ParentIdGuid filled), created when
-- the installment plan is approved, and marked paid like any challan. This is what the ERP's own installment list
-- (UNIStudentsInstallments_GetList) reads, so the tab reads the same:
--   Approved plans = installment challans that are not cancelled or expired (ChallanStatus not 3, 40, 63, 64).
--                    Month = the challan's due date, Amount = NetChallanAmt.
--                    Paid when ChallanStatus is 1 or 62; Overdue when its validity date (else due date) has passed;
--                    otherwise Upcoming.
--   Pending plans  = plans not approved yet (UNIFeeInstallments.InstallmentStatus not 2 or 40) have no challans, so
--                    their rows come from the plan lines (UNIFeeInstallmentsDetail): never Paid.
-- The date range is on the installment month.
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Installments;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Installments
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From  DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To    DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));
    DECLARE @Today DATE = CAST(GETDATE() AS DATE);

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    ),
    Ins AS (
        -- installments of approved plans: the installment challans
        SELECT DATEFROMPARTS(YEAR(ISNULL(C.DueDate, C.TransDate)), MONTH(ISNULL(C.DueDate, C.TransDate)), 1) AS [Month],
               C.DegreeId, C.ParentIdGuid AS PlanId,
               ISNULL(C.NetChallanAmt, ISNULL(C.TotalAmount, 0) - ISNULL(C.TotalDiscountAmt, 0)) AS Amount,
               CAST('Approved' AS VARCHAR(10)) AS PlanStatus,
               CAST(CASE WHEN C.ChallanStatus IN (1, 62) THEN 'Paid'
                         WHEN ISNULL(C.ValidityDate, C.DueDate) < @Today THEN 'Overdue'
                         ELSE 'Upcoming' END AS VARCHAR(10)) AS [State]
        FROM UNIStudentsChallan C WITH (NOLOCK)
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = C.StudentId AND S.Status = 1
        WHERE C.CompanyId = @CompanyId
          AND C.Status = 1
          AND C.ParentIdGuid IS NOT NULL
          AND C.ChallanStatus NOT IN (3, 40, 63, 64)
          AND (ISNULL(@BranchId, 0) = 0 OR C.BranchId = @BranchId)
          AND (@SemesterId IS NULL OR C.RegisterInId = @SemesterId)
          AND ISNULL(C.DueDate, C.TransDate) >= @From AND ISNULL(C.DueDate, C.TransDate) < @To

        UNION ALL

        -- installments of plans still waiting for approval: the plan lines
        SELECT DATEFROMPARTS(YEAR(D.[Month]), MONTH(D.[Month]), 1),
               S.DegreeId, H.Id, ISNULL(D.Amount, 0),
               CAST('Pending' AS VARCHAR(10)),
               CAST(CASE WHEN D.[Month] < @Today THEN 'Overdue' ELSE 'Upcoming' END AS VARCHAR(10))
        FROM UNIFeeInstallmentsDetail D WITH (NOLOCK)
        JOIN UNIFeeInstallments H WITH (NOLOCK) ON H.Id = D.FeeInstallmentId
        JOIN UNIStudents S WITH (NOLOCK) ON S.Id = H.StudentId AND S.Status = 1
        WHERE H.CompanyId = @CompanyId
          AND H.Status = 1 AND D.Status = 1
          AND ISNULL(H.InstallmentStatus, 0) NOT IN (2, 40)
          AND D.[Month] IS NOT NULL
          AND (ISNULL(@BranchId, 0) = 0 OR H.BranchId = @BranchId)
          AND (@SemesterId IS NULL OR H.SemesterId = @SemesterId)
          AND D.[Month] >= @From AND D.[Month] < @To
    )
    SELECT I.[Month], G.Id AS DegreeId, G.DegreeCode, G.DegreeName AS Degree, G.Department, G.Faculty,
           I.PlanStatus, I.[State],
           COUNT(DISTINCT I.PlanId) AS Plans,
           COUNT(*)                 AS Installments,
           SUM(I.Amount)            AS Amount
    FROM Ins I
    JOIN Deg G ON G.Id = I.DegreeId
    GROUP BY I.[Month], G.Id, G.DegreeCode, G.DegreeName, G.Department, G.Faculty, I.PlanStatus, I.[State]
    ORDER BY I.[Month], G.DegreeName;
END
GO

-- 09. Dashboard_MyUni_Discounts: fee discounts and scholarships per type, category, fee type, degree and status (Discounts tab).
-- One row of UNIFeeDiscountDetails is one discount given to a student on a fee type.
-- StatusText: Approved when the discount's DiscountStatus = 2, else Pending. The date range is on the discount start date.
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Discounts;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Discounts
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @DepartmentId INT              = NULL,   -- NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- NULL = all
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To   DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));

    -- Deg: the degree programs the filters allow, with their department and faculty names.
    ;WITH Deg AS (
        SELECT UD.Id, UD.DegreeCode, UD.DegreeName, UD.Status, D.[Description] AS Department, FA.Faculty
        FROM UNIDegree UD WITH (NOLOCK)
        LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = UD.DepartmentId
        OUTER APPLY (SELECT TOP 1 F.[Description] AS Faculty
                     FROM FacultyDetails FD WITH (NOLOCK)
                     JOIN Faculty F WITH (NOLOCK) ON F.FacultyGUID = FD.FacultyGUID
                     WHERE FD.DepartmentId = UD.DepartmentId
                     ORDER BY F.[Description]) FA
        WHERE UD.CompanyId = @CompanyId
          AND (@DegreeId IS NULL OR UD.Id = @DegreeId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR UD.DepartmentId = @DepartmentId)
          AND (@FacultyId IS NULL OR EXISTS (SELECT 1 FROM FacultyDetails FD2 WITH (NOLOCK)
                                             WHERE FD2.DepartmentId = UD.DepartmentId AND FD2.FacultyGUID = @FacultyId))
    )
    SELECT ISNULL(DT.[Description], 'Other discount') AS DiscountType,
           ISNULL(DC.[Description], 'No category')    AS Category,
           FT.[Description] AS FeeType,
           G.Id AS DegreeId, G.DegreeCode, G.DegreeName AS Degree, G.Department, G.Faculty,
           CASE WHEN H.DiscountStatus = 2 THEN 'Approved' ELSE 'Pending' END AS StatusText,
           COUNT(DISTINCT D.StudentId)         AS Students,
           SUM(ISNULL(D.DiscountedAmount, 0))  AS Amount
    FROM UNIFeeDiscountDetails D WITH (NOLOCK)
    JOIN UNIFeeDiscount H WITH (NOLOCK) ON H.Id = D.FeedDiscountId
    JOIN UNIStudents S WITH (NOLOCK) ON S.Id = D.StudentId AND S.Status = 1
    JOIN Deg G ON G.Id = ISNULL(D.DegreeId, S.DegreeId)
    LEFT JOIN UNIDiscountsTypes DT ON DT.Id = D.DiscountsTypeId
    LEFT JOIN UNIDiscountCatagory DC ON DC.Id = H.DiscountCatagoryId
    LEFT JOIN UNIFeeTypes FT WITH (NOLOCK) ON FT.Id = D.FeeTypeId
    WHERE H.CompanyId = @CompanyId
      AND H.Status = 1 AND D.Status = 1
      AND (ISNULL(@BranchId, 0) = 0 OR H.BranchId = @BranchId)
      AND (@SemesterId IS NULL OR H.AppliedSemesterId = @SemesterId)
      AND (H.StartDate IS NULL OR (H.StartDate >= @From AND H.StartDate < @To))
    GROUP BY DT.[Description], DC.[Description], FT.[Description],
             G.Id, G.DegreeCode, G.DegreeName, G.Department, G.Faculty,
             CASE WHEN H.DiscountStatus = 2 THEN 'Approved' ELSE 'Pending' END
    ORDER BY SUM(ISNULL(D.DiscountedAmount, 0)) DESC;
END

-- 10. Dashboard_MyUni_Employees: the staff of the company, one row per employee (Employees tab).
-- Every employee that is active, or that has left (so leavers can be counted). IsActive = active and not discontinued.
-- No personal details (CNIC, phone, address) are returned.
-- With dates chosen: the employees who worked at some point between @StartDate and @EndDate (hired on or before
-- @EndDate, and not gone before @StartDate), and IsActive says whether they were still employed on @EndDate.
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_Employees;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_Employees
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @DepartmentId INT              = NULL,   -- the employee's department (Employees.DepartmentId); NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To   DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));

    SELECT E.Id AS EmployeeId, E.Code, E.[Name],
           NULLIF(LTRIM(RTRIM(E.Designation)), '') AS Designation,
           D.[Description]  AS Department,
           ET.[Description] AS EmployeeType,
           NULLIF(LTRIM(RTRIM(E.EmployeesStatus)), '') AS EmploymentStatus,   -- Permanent, Contract, Daily Wages ...
           NULLIF(LTRIM(RTRIM(E.Gender)), '') AS Gender,
           B.[Description]  AS Branch,
           E.HiringDate, E.DiscontinuationDate,
           CAST(CASE WHEN (E.Status = 1 AND E.DiscontinuationDate IS NULL) OR E.DiscontinuationDate >= @To THEN 1 ELSE 0 END AS BIT) AS IsActive
    FROM Employees E WITH (NOLOCK)
    LEFT JOIN Departments D    WITH (NOLOCK) ON D.Id  = E.DepartmentId
    LEFT JOIN EmployeeTypes ET WITH (NOLOCK) ON ET.Id = E.TypeId
    LEFT JOIN Branches B       WITH (NOLOCK) ON B.Id  = E.BranchId
    WHERE E.CompanyId = @CompanyId
      AND (E.Status = 1 OR E.DiscontinuationDate IS NOT NULL)
      AND (E.HiringDate IS NULL OR E.HiringDate < @To)
      AND (E.DiscontinuationDate IS NULL OR E.DiscontinuationDate >= @From)
      AND (ISNULL(@BranchId, 0) = 0 OR E.BranchId = @BranchId)
      AND (ISNULL(@DepartmentId, 0) = 0 OR E.DepartmentId = @DepartmentId)
    ORDER BY E.[Name];
END

-- 11. Dashboard_MyUni_SalaryStructure: the current salary structure of each active employee, one row per employee (Salary structure tab).
-- The current structure is the active row of HRMSalaryStructure (Status = 1); when an employee has more than one,
-- the latest by effective date is used. Expired structures are left out.
-- With @EndDate chosen: the structure that was in force on that date (effective on or before it and not yet expired),
-- for employees who were employed then. @StartDate is not used: a salary structure is a state, not an event.
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_SalaryStructure;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_SalaryStructure
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @DepartmentId INT              = NULL,   -- the employee's department (Employees.DepartmentId); NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- the date the structure must be in force on; NULL = today
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @To DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));

    ;WITH Cur AS (
        SELECT S.EmployeeId, S.BasicSalary, S.SpecialAllowance, S.TotalAllowance, S.GrossSalary, S.RatePerDay, S.EffectiveDate,
               ROW_NUMBER() OVER (PARTITION BY S.EmployeeId ORDER BY S.EffectiveDate DESC, S.Id DESC) AS Rn
        FROM HRMSalaryStructure S WITH (NOLOCK)
        WHERE S.CompanyId = @CompanyId
          AND ((@EndDate IS NULL AND S.Status = 1)
               OR (@EndDate IS NOT NULL AND S.EffectiveDate < @To AND (S.ExpiryDate IS NULL OR S.ExpiryDate >= @To)))
    )
    SELECT E.Id AS EmployeeId, E.Code, E.[Name],
           NULLIF(LTRIM(RTRIM(E.Designation)), '') AS Designation,
           D.[Description] AS Department,
           NULLIF(LTRIM(RTRIM(E.EmployeesStatus)), '') AS EmploymentStatus,
           B.[Description] AS Branch,
           ISNULL(C.BasicSalary, 0)      AS BasicSalary,
           ISNULL(C.SpecialAllowance, 0) AS SpecialAllowance,
           ISNULL(C.TotalAllowance, 0)   AS TotalAllowance,
           ISNULL(C.GrossSalary, 0)      AS GrossSalary,
           ISNULL(C.RatePerDay, 0)       AS RatePerDay,
           C.EffectiveDate
    FROM Cur C
    JOIN Employees E WITH (NOLOCK) ON E.Id = C.EmployeeId
         AND ((E.Status = 1 AND E.DiscontinuationDate IS NULL) OR E.DiscontinuationDate >= @To)
         AND (E.HiringDate IS NULL OR E.HiringDate < @To)
    LEFT JOIN Departments D WITH (NOLOCK) ON D.Id = E.DepartmentId
    LEFT JOIN Branches B    WITH (NOLOCK) ON B.Id = E.BranchId
    WHERE C.Rn = 1
      AND (ISNULL(@BranchId, 0) = 0 OR E.BranchId = @BranchId)
      AND (ISNULL(@DepartmentId, 0) = 0 OR E.DepartmentId = @DepartmentId)
    ORDER BY ISNULL(C.GrossSalary, 0) DESC, E.[Name];
END

-- 12. Dashboard_MyUni_SalarySheet: payroll per month, department and branch (Salary sheet tab).
-- From the salary sheets (HRMSalarySheet, active rows) and their lines. The date range is on the sheet month.
-- Gross  = basic + special + total + monthly + recurring allowances.
-- NetPay = the formula of the ERP report HRMSalarySheet_Rp_GetByMonth, which adds up to the sheet's own NetAmount:
--          gross - (monthly and recurring deductions + absent deductions + loans + EOBI + income tax + provident fund).
-- The line columns GrossSalary and NetSalaryPayable are not used: they are empty or do not match the sheet total.
-- Sheets are counted whatever their approval status (SalarySheetStatus).
DROP PROCEDURE IF EXISTS dbo.Dashboard_MyUni_SalarySheet;
GO
CREATE PROCEDURE dbo.Dashboard_MyUni_SalarySheet
    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @DepartmentId INT              = NULL,   -- the employee's department (Employees.DepartmentId); NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To   DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));

    ;WITH L AS (
        SELECT DATEFROMPARTS(YEAR(S.[Month]), MONTH(S.[Month]), 1) AS [Month],
               S.Id AS SheetId, E.DepartmentId, ISNULL(D.BranchId, S.BranchId) AS BranchId, D.EmployeeId,
               ISNULL(D.BasicSalary, 0) AS Basic,
               ISNULL(D.SpecialAllowances, 0) + ISNULL(D.TotalAllowances, 0) + ISNULL(D.MonthlyAllowances, 0) + ISNULL(D.RecurringAllowances, 0) AS Allowances,
               ISNULL(D.OverTimeAmt, 0)   AS OverTime,
               ISNULL(D.ArrearsAmount, 0) AS Arrears,
               ISNULL(D.MonthlyDeductions, 0) + ISNULL(D.RecurringDeductions, 0) + ISNULL(D.AbsentDeductions, 0) AS Deductions,
               ISNULL(D.TotalLoans, 0)    AS Loans,
               ISNULL(D.IncomeTax, 0)     AS IncomeTax,
               ISNULL(D.EOBI, 0)          AS EOBI,
               ISNULL(D.ProvidentFund, 0) AS ProvidentFund
        FROM HRMSalarySheet S WITH (NOLOCK)
        JOIN HRMSalarySheetDetail D WITH (NOLOCK) ON D.HRMSalarySheetId = S.Id
        JOIN Employees E WITH (NOLOCK) ON E.Id = D.EmployeeId
        WHERE S.CompanyId = @CompanyId
          AND S.Status = 1
          AND S.[Month] >= @From AND S.[Month] < @To
          AND (ISNULL(@BranchId, 0) = 0 OR ISNULL(D.BranchId, S.BranchId) = @BranchId)
          AND (ISNULL(@DepartmentId, 0) = 0 OR E.DepartmentId = @DepartmentId)
    )
    SELECT L.[Month], DP.[Description] AS Department, B.[Description] AS Branch,
           COUNT(DISTINCT L.SheetId)    AS Sheets,
           COUNT(DISTINCT L.EmployeeId) AS Employees,
           SUM(L.Basic)         AS BasicSalary,
           SUM(L.Allowances)    AS Allowances,
           SUM(L.OverTime)      AS OverTime,
           SUM(L.Arrears)       AS Arrears,
           SUM(L.Basic + L.Allowances) AS GrossSalary,
           SUM(L.Deductions)    AS Deductions,
           SUM(L.Loans)         AS Loans,
           SUM(L.IncomeTax)     AS IncomeTax,
           SUM(L.EOBI)          AS EOBI,
           SUM(L.ProvidentFund) AS ProvidentFund,
           SUM((L.Basic + L.Allowances) - (L.Deductions + L.Loans + L.EOBI + L.IncomeTax + L.ProvidentFund)) AS NetPay
    FROM L
    LEFT JOIN Departments DP WITH (NOLOCK) ON DP.Id = L.DepartmentId
    LEFT JOIN Branches B     WITH (NOLOCK) ON B.Id  = L.BranchId
    GROUP BY L.[Month], DP.[Description], B.[Description]
    ORDER BY L.[Month], DP.[Description];
END

    @CompanyId    INT              = NULL,
    @BranchId     INT              = NULL,   -- NULL or 0 = all
    @FacultyId    UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @DepartmentId INT              = NULL,   -- the employee's department (Employees.DepartmentId); NULL or 0 = all
    @DegreeId     UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @SemesterId   UNIQUEIDENTIFIER = NULL,   -- not used by the HR tabs
    @StartDate    DATE             = NULL,   -- NULL = no limit
    @EndDate      DATE             = NULL    -- whole day included; NULL = no limit
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @From DATE = ISNULL(@StartDate, '19000101');
    DECLARE @To   DATE = DATEADD(DAY, 1, ISNULL(@EndDate, '99991230'));

    SELECT DATEFROMPARTS(YEAR(A.[Month]), MONTH(A.[Month]), 1) AS [Month],
           DP.[Description] AS Department, B.[Description] AS Branch,
           COUNT(DISTINCT D.EmployeeId)       AS Employees,
           SUM(ISNULL(D.NoOfDays, 0))         AS DaysWorked,
           SUM(ISNULL(D.Absenties, 0))        AS Absences,
           SUM(ISNULL(D.Leaves, 0))           AS Leaves,
           SUM(ISNULL(D.NoOfLateIn, 0))       AS LateArrivals,
           SUM(ISNULL(D.NoOfLateOut, 0))      AS EarlyDepartures,
           SUM(ISNULL(D.InMintLate, 0))       AS LateMinutes,
           SUM(ISNULL(D.OverTimeAmount, 0))   AS OverTimeAmount,
           SUM(ISNULL(D.DeductionAmount, 0))  AS DeductionAmount
    FROM HRMAttendenceApproval A WITH (NOLOCK)
    JOIN HRMAttendenceApprovalDetail D WITH (NOLOCK) ON D.AttendanceApprovalId = A.Id
    JOIN Employees E WITH (NOLOCK) ON E.Id = D.EmployeeId
    LEFT JOIN Departments DP WITH (NOLOCK) ON DP.Id = E.DepartmentId
    LEFT JOIN Branches B     WITH (NOLOCK) ON B.Id  = A.BranchId
    WHERE A.CompanyId = @CompanyId
      AND A.Status = 1
      AND A.[Month] >= @From AND A.[Month] < @To
      AND (ISNULL(@BranchId, 0) = 0 OR A.BranchId = @BranchId)
      AND (ISNULL(@DepartmentId, 0) = 0 OR E.DepartmentId = @DepartmentId)
    GROUP BY DATEFROMPARTS(YEAR(A.[Month]), MONTH(A.[Month]), 1), DP.[Description], B.[Description]
    ORDER BY 1, DP.[Description];
END
GO

SET NOEXEC OFF;
GO
